# Claude Actions Ring — план реализации

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** в Actions Ring мыши MX Master 4 появляется папка «Claude». В ней лежат открытые чаты Claude Code из VS Code. Клик по чату ведёт на следующий уровень с действиями: открыть чат, открыть все репо ветки в Rider, MR в GitLab, Jira, папки. Уровни сменяются «перезапуском кольца»: плагин переподписывает свои слоты и сам открывает кольцо заново.

**Architecture:** решение состоит из двух сборок.
- `ClaudeRing.Core` — чистая логика без зависимостей от Logi, покрыта unit-тестами:
  - чтение открытых вкладок Claude из state.vscdb VS Code;
  - заголовок и ветка чата из jsonl сессии;
  - поиск репо по ключу задачи;
  - ссылки;
  - машина состояний меню.
- `ClaudeRingPlugin` — плагин Logi Actions SDK (C#). В нём 8 команд-слотов с живыми подписями. Он вызывает Core и открывает кольцо заново через внутренний `RadialMenuMessageHandler.OnShowDevice(true)` службы LogiPluginService (рефлексия).

**Tech Stack:** C# / .NET (framework брать из шаблона `logiplugintool`), xUnit, System.Text.Json, `/usr/bin/sqlite3` CLI (без нативных NuGet-зависимостей), Logi Actions SDK (PluginApi.dll из `/Applications/Utilities/LogiPluginService.app/Contents/MonoBundle/`).

## Global Constraints

- **Платформа:** macOS, Logi Options+ 2.9.1013853, LogiPluginService 6.4.2.3414, MX Master 4.

- **Корни репо:** `~/Projects`. Раскладки `<repo>-<KEY>` (сейчас) и `<KEY>/<repo>` (новое правило).

- **Git:** репо локальное, без remote, ветка `main`. Пользователь разрешил коммитить после каждой задачи без отдельного вопроса. Сообщения коммитов на английском, одной строкой, без подписей. Не пушить.

```csharp
namespace Loupedeck.ClaudeRingPlugin
{
    using System;
    using System.Linq;
    using System.Reflection;
    using System.Threading.Tasks;

    // Opens the Actions Ring from inside LogiPluginService.
    // The service keeps its main object in a static field of ServiceTelemetrySender.
    // Field names are obfuscated, so we search by type.
    internal static class RingReopener
    {
        private static object _handler;

        public static void ShowAfter(Int32 delayMs) =>
            Task.Run(async () =>
            {
                await Task.Delay(delayMs);
                if (!TryShow(out var error))
                {
                    PluginLog.Error($"Кольцо не открылось: {error}");
                }
            });

        public static Boolean TryShow(out String error)
        {
            error = null;
            try
            {
                var handler = _handler ??= FindRadialMenuHandler(FindDeviceManager());
                var show = handler.GetType().GetMethod("OnShowDevice", new[] { typeof(Boolean) });
                show.Invoke(handler, new Object[] { true });
                return true;
            }
            catch (Exception ex)
            {
                _handler = null;
                error = ex.GetBaseException().Message;
                return false;
            }
        }

        internal static Object FindRadialMenuHandler(Object deviceManager) =>
            deviceManager.GetType().GetProperty("RadialMenuMessageHandler").GetValue(deviceManager);

        internal static Object FindDeviceManager()
        {
            var assembly = AppDomain.CurrentDomain.GetAssemblies()
                .First(a => a.GetName().Name == "LoupedeckService");
            var telemetry = assembly.GetType("Loupedeck.Service.ServiceTelemetrySender", throwOnError: true);
            var serviceField = telemetry
                .GetFields(BindingFlags.Static | BindingFlags.NonPublic | BindingFlags.Public)
                .First(f => f.FieldType.FullName == "Loupedeck.LoupedeckService");
            var service = serviceField.GetValue(null)
                ?? throw new InvalidOperationException("LoupedeckService is not created yet");
            return service.GetType().GetProperty("Loupedeck7DeviceManager").GetValue(service);
        }
    }
}
```

- [ ] **Step 2: RingWatcher**

```csharp
namespace Loupedeck.ClaudeRingPlugin
{
    using System;
    using System.Collections.Generic;
    using System.Reflection;
    using System.Threading;

    // Listens to ring events inside LogiPluginService.
    // Every event goes to the log. The event named resetEvent means "the ring was closed":
    // if the plugin did not close it on purpose, onClosedByUser runs.
    internal static class RingWatcher
    {
        private static readonly List<(Object Target, EventInfo Event, Delegate Handler)> Subscriptions = new();
        private static Int64 _expectReopenUntilTicks;
        private static String _resetEvent;
        private static Action _onClosedByUser;

        public static void Start(String resetEvent, Action onClosedByUser)
        {
            _resetEvent = resetEvent;
            _onClosedByUser = onClosedByUser;
            try
            {
                var deviceManager = RingReopener.FindDeviceManager();
                var handler = RingReopener.FindRadialMenuHandler(deviceManager);
                Subscribe(handler, "MouseEnter");
                Subscribe(handler, "MouseLeave");
                var device = deviceManager.GetType().GetProperty("RadialMenuDevice")?.GetValue(deviceManager);
                if (device != null) Subscribe(device, "DeviceRedrawRequested");
            }
            catch (Exception ex)
            {
                PluginLog.Error(ex, "Не удалось подписаться на события кольца, автосброса уровня не будет");
            }
        }

        public static void Stop()
        {
            foreach (var (target, evt, handler) in Subscriptions) evt.RemoveEventHandler(target, handler);
            Subscriptions.Clear();
        }

        // Call before the plugin re-opens the ring: the next close is ours, not the user's.
        public static void ExpectPluginReopen() =>
            Interlocked.Exchange(ref _expectReopenUntilTicks, DateTime.UtcNow.AddSeconds(3).Ticks);

        private static void Subscribe(Object target, String eventName)
        {
            var evt = target.GetType().GetEvent(eventName);
            if (evt == null)
            {
                PluginLog.Warning($"Событие {eventName} не найдено у {target.GetType().Name}");
                return;
            }
            var sink = new Sink(eventName);
            var handler = Delegate.CreateDelegate(evt.EventHandlerType, sink, typeof(Sink).GetMethod(nameof(Sink.Handle)));
            evt.AddEventHandler(target, handler);
            Subscriptions.Add((target, evt, handler));
        }

        private static void OnEvent(String name)
        {
            PluginLog.Info($"Событие кольца: {name}");
            if (String.IsNullOrEmpty(_resetEvent) || name != _resetEvent) return;
            if (Interlocked.Exchange(ref _expectReopenUntilTicks, 0) > DateTime.UtcNow.Ticks) return;
            _onClosedByUser?.Invoke();
        }

        // Binds to any EventHandler<T> because T derives from EventArgs.
        private sealed class Sink
        {
            private readonly String _name;
            public Sink(String name) => _name = name;
            public void Handle(Object sender, EventArgs e) => OnEvent(_name);
        }
    }
}
```

```csharp
namespace ClaudeRing.Core;

using System.Diagnostics;

public interface IProcessRunner
{
    // Runs a process and returns its stdout. Throws on non-zero exit code.
    string Run(string file, params string[] args);

    // Starts a process and does not wait for it.
    void Start(string file, params string[] args);
}

public sealed class ProcessRunner : IProcessRunner
{
    public string Run(string file, params string[] args)
    {
        using var p = Process.Start(Info(file, args))!;
        var output = p.StandardOutput.ReadToEnd();
        p.WaitForExit(5000);
        if (p.ExitCode != 0) throw new InvalidOperationException($"{file} exited with {p.ExitCode}: {p.StandardError.ReadToEnd()}");
        return output;
    }

    public void Start(string file, params string[] args) => Process.Start(Info(file, args))?.Dispose();

    private static ProcessStartInfo Info(string file, string[] args)
    {
        var info = new ProcessStartInfo(file) { RedirectStandardOutput = true, RedirectStandardError = true, UseShellExecute = false };
        foreach (var a in args) info.ArgumentList.Add(a);
        return info;
    }
}
```

```csharp
namespace ClaudeRing.Core;

using System.Text.Json;

public sealed class OpenChatsReader(string codeUserDir, string claudeIdeDir, IProcessRunner runner, Func<int, bool> isPidAlive)
{
    private const string StateKey = "Anthropic.claude-code";

    public IReadOnlyList<ChatTab> Read()
    {
        var live = LiveWindows();
        var result = new List<ChatTab>();
        var storage = Path.Combine(codeUserDir, "workspaceStorage");
        if (!Directory.Exists(storage)) return result;

        var dirs = Directory.GetDirectories(storage).Where(d => File.Exists(Path.Combine(d, "state.vscdb"))).ToList();
        foreach (var dir in dirs)
        {
            var folder = ReadFolder(dir);
            if (folder != null && live.Folders.Contains(folder)) result.AddRange(ReadTabs(dir, folder));
        }

        if (live.HasEmptyWindow)
        {
            var newestEmpty = dirs.Where(d => ReadFolder(d) == null)
                .OrderByDescending(d => File.GetLastWriteTimeUtc(Path.Combine(d, "state.vscdb")))
                .FirstOrDefault();
            if (newestEmpty != null) result.AddRange(ReadTabs(newestEmpty, null));
        }
        return Dedupe(result);
    }

    public static IReadOnlyList<ChatTab> ParsePanelTabs(string stateJson, string? folder)
    {
        using var doc = JsonDocument.Parse(stateJson);
        if (!doc.RootElement.TryGetProperty("panelTabSessions", out var tabs)) return [];
        return tabs.EnumerateArray()
            .Select(t => new ChatTab(t.GetProperty("sessionId").GetString()!, t.GetProperty("title").GetString() ?? "", folder))
            .ToList();
    }

    public static IReadOnlyList<ChatTab> Dedupe(IEnumerable<ChatTab> tabs) =>
        tabs.GroupBy(t => t.SessionId).Select(g => g.First()).ToList();

    private IEnumerable<ChatTab> ReadTabs(string dir, string? folder)
    {
        var db = Path.Combine(dir, "state.vscdb");
        var json = runner.Run("/usr/bin/sqlite3", "-readonly", "-json", db,
            $"select value from ItemTable where key='{StateKey}'");
        if (string.IsNullOrWhiteSpace(json)) return [];
        using var rows = JsonDocument.Parse(json);
        var value = rows.RootElement.EnumerateArray().Select(r => r.GetProperty("value").GetString()).FirstOrDefault();
        return value == null ? [] : ParsePanelTabs(value, folder);
    }

    private static string? ReadFolder(string dir)
    {
        var file = Path.Combine(dir, "workspace.json");
        if (!File.Exists(file)) return null;
        using var doc = JsonDocument.Parse(File.ReadAllText(file));
        if (!doc.RootElement.TryGetProperty("folder", out var uri)) return null;
        return Uri.TryCreate(uri.GetString(), UriKind.Absolute, out var u) && u.IsFile ? u.LocalPath.TrimEnd('/') : null;
    }

    private (HashSet<string> Folders, bool HasEmptyWindow) LiveWindows()
    {
        var folders = new HashSet<string>();
        var hasEmpty = false;
        if (!Directory.Exists(claudeIdeDir)) return (folders, hasEmpty);
        foreach (var file in Directory.GetFiles(claudeIdeDir, "*.lock"))
        {
            using var doc = JsonDocument.Parse(File.ReadAllText(file));
            if (!doc.RootElement.TryGetProperty("pid", out var pid) || !isPidAlive(pid.GetInt32())) continue;
            var list = doc.RootElement.TryGetProperty("workspaceFolders", out var wf) ? wf.EnumerateArray().Select(x => x.GetString()!.TrimEnd('/')).ToList() : [];
            if (list.Count == 0) hasEmpty = true;
            folders.UnionWith(list);
        }
        return (folders, hasEmpty);
    }

    public static bool IsPidAlive(int pid)
    {
        try { return !System.Diagnostics.Process.GetProcessById(pid).HasExited; }
        catch (ArgumentException) { return false; }
    }
}
```

```csharp
public sealed class SessionInfoReader(string projectsDir)
{
    public SessionInfo Read(string sessionId)
    {
        if (!Directory.Exists(projectsDir)) return new(null, null, null, null);
        var file = Directory.EnumerateFiles(projectsDir, sessionId + ".jsonl", SearchOption.AllDirectories).FirstOrDefault();
        if (file == null) return new(null, null, null, null);

        string? custom = null, ai = null, branch = null, cwd = null;
        foreach (var line in File.ReadLines(file))
        {
            if (!line.StartsWith('{')) continue;
            try
            {
                using var doc = JsonDocument.Parse(line);
                var r = doc.RootElement;
                custom = Str(r, "customTitle") ?? custom;
                ai = Str(r, "aiTitle") ?? ai;
                branch = Str(r, "gitBranch") ?? branch;
                cwd = Str(r, "cwd") ?? cwd;
            }
            catch (JsonException) { }
        }
        return new(custom, ai, branch, cwd);
    }

    private static string? Str(JsonElement e, string name) =>
        e.TryGetProperty(name, out var v) && v.ValueKind == JsonValueKind.String && v.GetString() is { Length: > 0 } s ? s : null;
```

```csharp
namespace ClaudeRing.Core;

using System.Text.RegularExpressions;

public sealed class RepoFinder(IEnumerable<string> roots)
{
    public IReadOnlyList<RepoRef> Find(string key)
    {
        var branchMatch = new Regex($@"(^|[/_-]){Regex.Escape(key)}($|[/_-])", RegexOptions.IgnoreCase);
        var result = new List<RepoRef>();
        foreach (var root in roots.Where(Directory.Exists))
        {
            foreach (var dir in Candidates(root))
            {
                var branch = ReadBranch(dir);
                if (branch != null && branchMatch.IsMatch(branch)) result.Add(new RepoRef(Path.GetFileName(dir), dir, branch));
            }
        }
        return result.OrderBy(r => r.Name, StringComparer.OrdinalIgnoreCase).ToList();
    }

    // Repos at depth 1 (<repo>-<KEY>) and depth 2 (<KEY>/<repo>).
    private static IEnumerable<string> Candidates(string root)
    {
        foreach (var d1 in Directory.EnumerateDirectories(root))
        {
            if (Path.GetFileName(d1).StartsWith('_') || Path.GetFileName(d1).StartsWith('.')) continue;
            if (IsRepo(d1)) { yield return d1; continue; }
            foreach (var d2 in Directory.EnumerateDirectories(d1).Where(IsRepo)) yield return d2;
        }
    }

    private static bool IsRepo(string dir) => Path.Exists(Path.Combine(dir, ".git"));

    private static string? ReadBranch(string repo)
    {
        var git = Path.Combine(repo, ".git");
        var gitDir = Directory.Exists(git) ? git : ParseGitDir(git, repo);
        var head = gitDir == null ? null : Path.Combine(gitDir, "HEAD");
        if (head == null || !File.Exists(head)) return null;
        var text = File.ReadAllText(head).Trim();
        return text.StartsWith("ref: refs/heads/") ? text["ref: refs/heads/".Length..] : null;
    }

    private static string? ParseGitDir(string gitFile, string repo)
    {
        var line = File.ReadAllText(gitFile).Trim();
        if (!line.StartsWith("gitdir: ")) return null;
        var path = line["gitdir: ".Length..];
        return Path.IsPathRooted(path) ? path : Path.GetFullPath(Path.Combine(repo, path));
    }
}
```

### Task 6: Машина состояний меню

**Уровни (N = `SlotCount`, последний слот на не-корневых уровнях = «← назад»):**

| Уровень | Слоты |
|---|---|
| Root | чаты по `Label`; если чатов > N, то N−1 чатов + «ещё ›» (листает страницы) |
| Chat | «Чат в VS Code», «Rider: все репо (k)», «MR ›», «Jira», «Репо ›», …, «← назад» |
| Mr | «Все MR», цели из `MrTargets` («→ Dev», «→ RC», «→ master»), «← назад» |
| Repos | по репо (клик → Rider только с ним), «Все в Finder», «← назад»; если репо > N−2, лишние не показываются (YAGNI) |

```csharp
namespace ClaudeRing.Core;

public abstract record MenuAction
{
    public sealed record FocusChat(ChatTab Tab) : MenuAction;
    public sealed record OpenRider(IReadOnlyList<RepoRef> Repos) : MenuAction;
    public sealed record OpenUrl(string Url) : MenuAction;
    public sealed record Reveal(IReadOnlyList<string> Paths) : MenuAction;
}

public sealed record ClickResult(MenuAction? Action, bool Reopen);

public sealed record SlotView(string Label, bool Enabled);

public sealed class RingMenu(IChatSource source, Links links, RingConfig config)
{
    private enum Level { Root, Chat, Mr, Repos }

    // One ring item: what it shows and what a click does.
    private sealed record Item(string Label, Func<ClickResult> OnClick);

    private const string Back = "← назад";
    private const string More = "ещё ›";

    private readonly object _lock = new();
    private IReadOnlyList<ChatContext> _chats = [];
    private Level _level = Level.Root;
    private ChatContext? _chat;
    private int _page;

    private int N => config.SlotCount;

    public void Reset()
    {
        lock (_lock)
        {
            _chats = source.Load();
            _level = Level.Root;
            _chat = null;
            _page = 0;
        }
    }

    public IReadOnlyList<SlotView> Render()
    {
        lock (_lock)
        {
            var items = Items();
            return Enumerable.Range(0, N)
                .Select(i => i < items.Count && items[i] != null ? new SlotView(items[i]!.Label, true) : new SlotView("", false))
                .ToList();
        }
    }

    public ClickResult Click(int slot)
    {
        Item? item;
        lock (_lock)
        {
            var items = Items();
            item = slot < items.Count ? items[slot] : null;
        }
        if (item == null) return new ClickResult(null, false);

        var result = item.OnClick();
        if (!result.Reopen) Reset();
        return result;
    }

    // Builds the slot list for the current level. Null means an empty slot.
    private List<Item?> Items() => _level switch
    {
        Level.Root => RootItems(),
        Level.Chat => WithBack(ChatItems(_chat!)),
        Level.Mr => WithBack(MrItems(_chat!)),
        Level.Repos => WithBack(RepoItems(_chat!)),
        _ => [],
    };

    private List<Item?> RootItems()
    {
        var perPage = _chats.Count > N ? N - 1 : N;
        var pageChats = _chats.Skip(_page * perPage).Take(perPage)
            .Select(c => (Item?)new Item(c.Label, () => Go(Level.Chat, c))).ToList();
        if (_chats.Count > N)
        {
            while (pageChats.Count < N - 1) pageChats.Add(null);
            var pages = (int)Math.Ceiling(_chats.Count / (double)perPage);
            pageChats.Add(new Item(More, () => { _page = (_page + 1) % pages; return new ClickResult(null, true); }));
        }
        return pageChats;
    }

    private List<Item> ChatItems(ChatContext c)
    {
        var items = new List<Item> { new("Чат в VS Code", () => Done(new MenuAction.FocusChat(c.Tab))) };
        if (c.TaskKey == null) return items;
        if (c.Repos.Count > 0) items.Add(new($"Rider: все репо ({c.Repos.Count})", () => Done(new MenuAction.OpenRider(c.Repos))));
        items.Add(new("MR ›", () => Go(Level.Mr, c)));
        if (links.JiraIssue(c.TaskKey) is { } jira) items.Add(new("Jira", () => Done(new MenuAction.OpenUrl(jira))));
        if (c.Repos.Count > 0) items.Add(new("Репо ›", () => Go(Level.Repos, c)));
        return items;
    }

    private List<Item> MrItems(ChatContext c)
    {
        var items = new List<Item> { new("Все MR", () => Done(new MenuAction.OpenUrl(links.GroupMrs(c.TaskKey!)))) };
        items.AddRange(config.MrTargets.Select(t => new Item(t.Label, () => Done(new MenuAction.OpenUrl(links.GroupMrs(c.TaskKey!, t.Branch))))));
        return items;
    }

    private List<Item> RepoItems(ChatContext c)
    {
        var items = c.Repos.Take(N - 2).Select(r => new Item(r.Name, () => Done(new MenuAction.OpenRider([r])))).ToList();
        items.Add(new("Все в Finder", () => Done(new MenuAction.Reveal(c.Repos.Select(r => r.Path).ToList()))));
        return items;
    }

    // Puts "back" into the last slot, the rest stays in order.
    private List<Item?> WithBack(List<Item> items)
    {
        var slots = items.Take(N - 1).Cast<Item?>().ToList();
        while (slots.Count < N - 1) slots.Add(null);
        slots.Add(new Item(Back, GoBack));
        return slots;
    }

    private ClickResult Go(Level level, ChatContext chat)
    {
        _level = level;
        _chat = chat;
        return new ClickResult(null, true);
    }

    private ClickResult GoBack()
    {
        _level = _level == Level.Chat ? Level.Root : Level.Chat;
        return new ClickResult(null, true);
    }

    private static ClickResult Done(MenuAction action) => new(action, false);
}
```

```csharp
namespace Loupedeck.ClaudeRingPlugin
{
    using System;
    using System.Threading;
    using ClaudeRing.Core;

    public class ClaudeRingPlugin : Plugin
    {
        public override Boolean UsesApplicationApiOnly => true;
        public override Boolean HasNoApplication => true;

        internal static RingMenu Menu { get; private set; }
        internal static RingConfig Config { get; private set; }
        internal static event Action SlotsChanged;

        private static Launcher _launcher;

        public ClaudeRingPlugin()
        {
            PluginLog.Init(this.Log);
            PluginResources.Init(this.Assembly);
        }

        public override void Load()
        {
            Config = RingConfig.Load(RingConfig.DefaultPath);
            var runner = new ProcessRunner();
            _launcher = new Launcher(runner, Config, Thread.Sleep);
            Menu = new RingMenu(ChatSource.CreateDefault(Config, runner), new Links(Config), Config);
            Refresh();
            RingWatcher.Start(Config.ResetOnRingEvent, onClosedByUser: Refresh);
        }

        public override void Unload() => RingWatcher.Stop();

        // Reads open chats again and shows the root level.
        internal static void Refresh()
        {
            try { Menu.Reset(); }
            catch (Exception ex) { PluginLog.Error(ex, "Не удалось прочитать открытые чаты"); }
            SlotsChanged?.Invoke();
        }

        internal static void Execute(MenuAction action)
        {
            try
            {
                switch (action)
                {
                    case MenuAction.FocusChat a: _launcher.FocusChat(a.Tab); break;
                    case MenuAction.OpenRider a: _launcher.OpenInRider(a.Repos); break;
                    case MenuAction.OpenUrl a: _launcher.OpenUrl(a.Url); break;
                    case MenuAction.Reveal a: _launcher.Reveal(a.Paths); break;
                }
            }
            catch (Exception ex)
            {
                PluginLog.Error(ex, $"Действие не выполнилось: {action}");
            }
        }
    }
}
```

```csharp
namespace Loupedeck.ClaudeRingPlugin
{
    using System;
    using System.Threading.Tasks;

    // Fixed ring slots. Labels come from RingMenu and change when the level changes.
    public class SlotCommand : PluginDynamicCommand
    {
        private const Int32 MaxSlots = 9;

        public SlotCommand()
        {
            for (var i = 0; i < MaxSlots; i++)
            {
                this.AddParameter(i.ToString(), $"Claude слот {i + 1}", "Claude Ring");
            }
            ClaudeRingPlugin.SlotsChanged += () => this.ActionImageChanged();
        }

        protected override String GetCommandDisplayName(String actionParameter, PluginImageSize imageSize)
        {
            var slots = ClaudeRingPlugin.Menu?.Render();
            var i = Int32.Parse(actionParameter);
            return slots != null && i < slots.Count ? slots[i].Label : "";
        }

        protected override void RunCommand(String actionParameter)
        {
            var result = ClaudeRingPlugin.Menu.Click(Int32.Parse(actionParameter));
            this.ActionImageChanged();
            if (result.Action != null)
            {
                Task.Run(() => ClaudeRingPlugin.Execute(result.Action));
            }
            if (result.Reopen && ClaudeRingPlugin.Config.ReopenDelayMs >= 0)
            {
                // The ring closes after this click; do not treat that close as "user closed it".
                RingWatcher.ExpectPluginReopen();
                RingReopener.ShowAfter(ClaudeRingPlugin.Config.ReopenDelayMs);
            }
        }
    }
}
```
